cmake_minimum_required(VERSION 3.24)
project(MyWallpaperAudioSystem VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/manifest.json" MW_MANIFEST)
configure_file(include/manifest_data.hpp.in manifest_data.hpp @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/contracts/audio-system/v1/schema.json" MW_SERVICE_SCHEMA)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/contracts/audio-system/v1/examples.json" MW_SERVICE_EXAMPLES)
# Keep each source literal below MSVC's 16 KiB limit; adjacent literals retain
# the exact public UTF-8 bytes, including newlines, without a parser dependency.
string(REPLACE "\n" "\n)A18SCHEMA\"\nR\"A18SCHEMA(" MW_SERVICE_SCHEMA "${MW_SERVICE_SCHEMA}")
string(REPLACE "\n" "\n)A18EXAMPLE\"\nR\"A18EXAMPLE(" MW_SERVICE_EXAMPLES "${MW_SERVICE_EXAMPLES}")
configure_file(include/service_contract_data.hpp.in service_contract_data.hpp @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/native/dependencies.lock.json" dependency_lock)
foreach(key path noticePath)
  string(JSON dep_path GET "${dependency_lock}" json ${key})
  if(key STREQUAL "path")
    string(JSON expected GET "${dependency_lock}" json sha256)
  else()
    string(JSON expected GET "${dependency_lock}" json noticeSha256)
  endif()
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${dep_path}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Locked dependency changed: ${dep_path}")
  endif()
endforeach()
add_library(audio_model STATIC src/model.cpp src/wire.cpp src/endpoint_selection.cpp)
target_include_directories(audio_model PUBLIC include third_party "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(audio_model PUBLIC /utf-8 /W4 /permissive- /EHsc /Brepro)
else()
  target_compile_options(audio_model PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_library(audio_service STATIC src/service_provider.cpp src/service_schema.cpp src/service_hash.cpp)
target_include_directories(audio_service PUBLIC include third_party "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(audio_service PUBLIC /utf-8 /W4 /permissive- /EHsc /Brepro)
else()
  target_compile_options(audio_service PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(service_provider_tests tests/service_provider_tests.cpp)
target_link_libraries(service_provider_tests PRIVATE audio_service)
add_test(NAME native_service_provider COMMAND service_provider_tests "${CMAKE_CURRENT_BINARY_DIR}/provider-transcript.json")
add_executable(model_tests tests/model_tests.cpp)
target_link_libraries(model_tests PRIVATE audio_model)
add_test(NAME audio_contract COMMAND model_tests)
add_executable(endpoint_selection_tests tests/endpoint_selection_tests.cpp)
target_link_libraries(endpoint_selection_tests PRIVATE audio_model)
add_test(NAME endpoint_selection COMMAND endpoint_selection_tests)
if(WIN32)
  if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 MSVC is required")
  endif()
  add_library(audio_windows STATIC src/audio.cpp src/media.cpp)
  target_link_libraries(audio_windows PUBLIC audio_model audio_service ole32 uuid windowsapp)
  target_compile_definitions(audio_windows PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_executable(companion src/main.cpp src/panel.cpp)
  target_link_libraries(companion PRIVATE audio_windows user32 gdi32 comctl32 shell32 shcore winmm)
  add_executable(audio_fixture tests/audio_fixture.cpp)
  target_link_libraries(audio_fixture PRIVATE audio_windows)
  add_executable(companion_fixture tests/companion_fixture.cpp)
  target_link_libraries(companion_fixture PRIVATE audio_windows user32)
  add_dependencies(companion_fixture companion)
  add_executable(service_companion_fixture tests/service_companion_fixture.cpp)
  target_link_libraries(service_companion_fixture PRIVATE audio_model audio_service user32)
  target_compile_definitions(service_companion_fixture PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_dependencies(service_companion_fixture companion)
  add_test(NAME hidden_native_service_provider COMMAND service_companion_fixture "$<TARGET_FILE:companion>" "${CMAKE_CURRENT_BINARY_DIR}/hidden-provider-transcript.json")
  set_tests_properties(hidden_native_service_provider PROPERTIES TIMEOUT 30)
  add_test(NAME private_companion_protocol COMMAND companion_fixture "$<TARGET_FILE:companion>")
  set_tests_properties(private_companion_protocol PROPERTIES TIMEOUT 30)
  add_test(NAME owned_silent_audio COMMAND audio_fixture)
  set_tests_properties(owned_silent_audio PROPERTIES TIMEOUT 30)
endif()
if(MW_NATIVE_DEVELOPMENT_MODULE)
  include("${MW_NATIVE_DEVELOPMENT_MODULE}")
  if(WIN32)
    mw_native_development(UNIT_TARGETS model_tests service_provider_tests endpoint_selection_tests UNIT_TESTS audio_contract native_service_provider endpoint_selection RUNTIME_TARGETS companion FIXTURE_TARGETS audio_fixture companion_fixture service_companion_fixture FIXTURE_TESTS owned_silent_audio private_companion_protocol hidden_native_service_provider)
  else()
    mw_native_development(UNIT_TARGETS model_tests service_provider_tests endpoint_selection_tests UNIT_TESTS audio_contract native_service_provider endpoint_selection)
  endif()
endif()
